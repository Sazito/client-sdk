## Runtime model

- Use `app/lib` for shared SDK setup.
- Use Client Components for browser interactions and token lifecycle.
- Use Server Components for read-heavy rendering.
- Use Server Actions for authenticated mutations.

## 1. Shared SDK files

```ts
// app/lib/sazito-client.ts
import { createSazitoClient } from '@sazito/client-sdk';

export const sazitoClient = createSazitoClient({
  domain: process.env.NEXT_PUBLIC_SAZITO_DOMAIN!,
  timeout: 15000,
});
```

```ts
// app/lib/sazito-server.ts
import { createSazitoClient } from '@sazito/client-sdk';

export function getSazitoServerClient() {
  return createSazitoClient({
    domain: process.env.SAZITO_STORE_DOMAIN!,
    timeout: 15000,
  });
}
```

## 2. Server Component example (parallel reads)

```tsx
// app/[lang]/(store)/page.tsx
import { getSazitoServerClient } from '@/app/lib/sazito-server';

export default async function StorePage() {
  const client = getSazitoServerClient();

  const [productsRes, categoriesRes] = await Promise.all([
    client.products.list({ page: 1, pageSize: 12, sort: 'newest' }),
    client.categories.list({ page: 1, pageSize: 20 }),
  ]);

  if (productsRes.error || categoriesRes.error) {
    return <p>Failed to load storefront data.</p>;
  }

  return (
    <section>
      <h1>Storefront</h1>
      <p>Products: {productsRes.data?.items.length ?? 0}</p>
      <p>Categories: {categoriesRes.data?.categories.length ?? 0}</p>
    </section>
  );
}
```

## 3. Client Component example (interactive query)

```tsx
// app/[lang]/(store)/search-box.tsx
'use client';

import { useState } from 'react';
import { sazitoClient } from '@/app/lib/sazito-client';

export function SearchBox() {
  const [query, setQuery] = useState('');
  const [results, setResults] = useState<any[]>([]);
  const [error, setError] = useState<string | null>(null);

  async function runSearch() {
    setError(null);

    const res = await sazitoClient.search.query(query, { page: 1, pageSize: 10 }, { timeout: 8000 });
    if (res.error) {
      setError(res.error.message);
      return;
    }

    setResults(res.data?.products.items ?? []);
  }

  return (
    <div>
      <input value={query} onChange={(e) => setQuery(e.target.value)} placeholder="Search products" />
      <button onClick={runSearch}>Search</button>
      {error ? <p>{error}</p> : null}
      <ul>{results.map((item) => <li key={item.url}>{item.name}</li>)}</ul>
    </div>
  );
}
```

## 4. Authenticated Server Action example

```ts
// app/[lang]/account/wallet/actions.ts
'use server';

import { cookies } from 'next/headers';
import { getSazitoServerClient } from '@/app/lib/sazito-server';

export async function getWalletBalanceAction() {
  const token = (await cookies()).get('user_id_token')?.value;
  const client = getSazitoServerClient();

  const res = await client.wallet.getBalance({
    headers: token ? { Authorization: token } : undefined,
    cache: false,
  });

  if (res.error) {
    throw new Error(res.error.message);
  }

  return res.data;
}
```

## 5. Usage in a Server Component

```tsx
// app/[lang]/account/wallet/page.tsx
import { getWalletBalanceAction } from './actions';

export default async function WalletPage() {
  const wallet = await getWalletBalanceAction();
  return <pre>{JSON.stringify(wallet, null, 2)}</pre>;
}
```

## 6. Drop-in checkout component

For a complete cart-to-payment flow, use `SazitoCheckoutPage` on a dedicated
client-facing route. Install the checkout package and its React peers:

```bash
pnpm add @sazito/client-sdk @sazito/checkout react@18 react-dom@18
```

Create a Client Component that initializes one SDK client, provides it to the
checkout, and passes the cart credentials to the page:

```tsx
// app/checkout/checkout-client.tsx
'use client';

import { createSazitoClient } from '@sazito/client-sdk';
import {
  SazitoCheckoutPage,
  SazitoProvider,
} from '@sazito/checkout/next';

const sazito = createSazitoClient({
  domain: process.env.NEXT_PUBLIC_SAZITO_DOMAIN!,
});

export function CheckoutClient({ cartIdentifier }: { cartIdentifier: string }) {
  return (
    <SazitoProvider client={sazito}>
      <SazitoCheckoutPage
        credentials={{ cart: { identifier: cartIdentifier } }}
        config={{
          locale: 'fa',
          continueShoppingUrl: '/',
        }}
      />
    </SazitoProvider>
  );
}
```

Render `CheckoutClient` from the route after reading the cart identifier from
your application state, cookie, or URL. Import `@sazito/checkout/styles.css`
once from the root layout, and see the
[Checkout guide](/docs/guides/checkout) for provider setup, theming, payment
returns, and custom button rendering.

## 7. Checklist

- Keep `NEXT_PUBLIC_SAZITO_DOMAIN` for browser code and `SAZITO_STORE_DOMAIN` for server code.
- Use Server Components for initial page payload and SEO-critical data.
- Use Server Actions for authenticated writes and sensitive flows.
- Always branch on `res.error` before using `res.data`.
